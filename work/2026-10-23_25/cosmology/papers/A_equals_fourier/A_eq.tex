\noindent\textbf{\large Our model $A(\hat n)$ = the Fourier fit to Migkas et al.\ (2020)}\\[4pt]
Sources on the vision shell, $d=r$; \ $\mu=\cos b_0\cos(\ell-\ell_0)$; \ $\kappa=\dfrac{GM(<D)}{H_0^2D^3}$; \ $W_m=\dfrac{\sin(m\pi/4)}{m\pi/4}$ (90$^\circ$ sectors).
\begin{equation*}
\underbrace{N\!\left[1+\alpha\kappa\,(3\mu^2-1)+\alpha\kappa\,\frac{r}{D}\Big(\tfrac92\mu-\tfrac{15}{2}\mu^3\Big)\right]}_{\text{our model}}
=\underbrace{a_0+\sum_{m=1}^{3}\big[c_m\cos m\ell+s_m\sin m\ell\big]}_{\text{Fourier fit}}
\end{equation*}
\begin{equation*}
A_{\rm fit}(\ell)=0.9965+0.0296\cos\ell+0.0434\sin\ell+0.0636\cos2\ell+0.0366\sin2\ell-0.0299\cos3\ell-0.0114\sin3\ell,\quad R^2=0.915
\end{equation*}
\textbf{Term by term} (fit projected on $\cos m(\ell-\ell_0)$, $\ell_0=15^\circ$, $c\equiv\cos b_0$):
\begin{align*}
m=0:&\quad 0.9965 = N\big[1+\alpha\kappa(\tfrac32c^2-1)\big]\\
m=1:&\quad +0.0398 = N\alpha\kappa\,\tfrac{r}{D}\,c\Big(\tfrac92-\tfrac{45}{8}c^2\Big)W_1\\
m=2:&\quad +0.0733 = N\alpha\kappa\,\tfrac32c^2\,W_2\\
m=3:&\quad -0.0292 = -N\alpha\kappa\,\tfrac{r}{D}\,\tfrac{15}{8}c^3\,W_3
\end{align*}
\textbf{Solution:} \ $c^2=0.695\ (|b_0|=33.5^\circ)$, \quad $\dfrac{r}{D}=0.812\ \Rightarrow\ \dfrac{D}{r}=1.23,\ D\approx17.6$ Gpc, \quad $N=0.992$, \quad $\alpha\kappa=0.111$.\\[4pt]
\textbf{Not reproducible by an axisymmetric model} (the $\sin m(\ell-15^\circ)$ parts): \ $m_1{:}\ {+0.034}$, \ $m_2{:}\ {-0.000}$, \ $m_3{:}\ {+0.013}$.\\[2pt]
The analytic form is a small-$r/D$ expansion evaluated at $r/D\approx0.8$. The full numeric model gives $R^2=0.776$ at $D/r\approx1.32$.
